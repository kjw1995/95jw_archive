{{/* hextra 테마의 components/last-updated.html 오버라이드. 페이지 하단에 작성일(frontmatter date)과 최종 수정일(git 커밋 날짜)을 함께 보여준다. 두 날짜가 같은 날이면 작성일만 표시한다. */}} {{- $created := .Date -}} {{- $updated := .Lastmod -}} {{- $displayUpdated := site.Params.displayUpdatedDate -}} {{- $hasCreated := not $created.IsZero -}} {{- $hasUpdated := and $displayUpdated (not $updated.IsZero) -}} {{/* 같은 날짜면 최종 수정일은 생략 */}} {{- if and $hasCreated $hasUpdated -}} {{- if eq (time.Format "2006-01-02" $created) (time.Format "2006-01-02" $updated) -}} {{- $hasUpdated = false -}} {{- end -}} {{- end -}} {{- if or $hasCreated $hasUpdated -}}
{{- if $hasCreated -}} 작성일 {{ partial "utils/format-date" $created }} {{- end -}} {{- if and $hasCreated $hasUpdated }} · {{ end -}} {{- if $hasUpdated -}} 최종 수정일 {{ partial "utils/format-date" $updated }} {{- end -}}

{{- end -}}
